# 649. Dota2 Senate

from collections import deque

class Solution:
    def predictPartyVictory(self, senate: str) -> str:
        # We want to treat the senate as a queue
        n = len(senate)
        r_queue = deque([i for i, s in enumerate(senate) if s == "R"])
        d_queue = deque([i for i, s in enumerate(senate) if s == "D"])

        while r_queue and d_queue:
            r_idx = r_queue.popleft()
            d_idx = d_queue.popleft()

            # The senator with the smaller index bans the other
            if r_idx < d_idx:
                # Radiant acts
                r_queue.append(r_idx + n)
            else:  # Dire acts
                d_queue.append(d_idx + n)
        
        return "Radiant" if r_queue else "Dire"


# Greedy algorithm -- cyclic reduction
# RT -> O(N)
# Space -> O(N)
