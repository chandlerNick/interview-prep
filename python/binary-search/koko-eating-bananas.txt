# 875. Koko Eating Bananas

class Solution:
    def minEatingSpeed(self, piles: List[int], h: int) -> int:

        low, high = 1, max(piles)
        
        while low < high:
            k = low + (high - low) // 2
            total_hours = 0
            for pile in piles:
                total_hours += (pile + k - 1) // k
            if total_hours <= h:
                high = k
            else:
                low = k + 1
            
        return low

RT -> O(nlog(max(P)))
Space -> O(1)
