# 162. Find Peak Element

class Solution:

    def findPeakElement(self, nums: List[int]) -> int:
        low, high = 0, len(nums) - 1
        
        while low < high:
            mid = low + (high - low) // 2
            
            if nums[mid] > nums[mid + 1]:
                high = mid
            else:
                low = mid + 1
                
        return low # low and high converge on the result

RT -> O(lgn)
Space -> O(1)
