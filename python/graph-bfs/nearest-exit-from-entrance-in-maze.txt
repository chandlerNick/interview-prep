# 1926. Nearest Exit from Entrance in Maze

from collections import deque

class Solution:
    def nearestExit(self, maze: List[List[str]], entrance: List[int]) -> int:
        rows, cols = len(maze), len(maze[0])
        start_r, start_c = entrance
        
        # Use a queue for (row, col, current_distance)
        queue = deque([(start_r, start_c, 0)])
        # Mark entrance as visited by changing it to a wall or using a set
        maze[start_r][start_c] = '+' 
        
        while queue:
            r, c, d = queue.popleft()
            
            for dr, dc in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
                nr, nc = r + dr, c + dc
                
                # 1. Check bounds and if it's an empty path
                if 0 <= nr < rows and 0 <= nc < cols and maze[nr][nc] == '.':
                    
                    # 2. Check if this path is an exit (on the border)
                    if nr == 0 or nr == rows - 1 or nc == 0 or nc == cols - 1:
                        return d + 1
                    
                    # 3. Mark as visited and continue BFS
                    maze[nr][nc] = '+'
                    queue.append((nr, nc, d + 1))
        
        return -1

RT -> O(MN)
Space -> O(MN)
