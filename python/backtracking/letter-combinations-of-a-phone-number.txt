# 17. Letter Combinations of a Phone Number

class Solution:
    def letterCombinations(self, digits: str) -> List[str]:
        n = len(digits)
        dtol = {
            '2' : ['a', 'b', 'c'],
            '3' : ['d', 'e', 'f'],
            '4' : ['g', 'h', 'i'],
            '5' : ['j', 'k', 'l'],
            '6' : ['m', 'n', 'o'],
            '7' : ['p', 'q', 'r', 's'],
            '8' : ['t', 'u', 'v'],
            '9' : ['w', 'x', 'y', 'z']
        }
        if n == 1:
            return dtol[digits]
        elif n == 2:
            return [
                    ''.join([i,j])
                    for i in dtol[digits[0]] 
                    for j in dtol[digits[1]]
                ]
        elif n == 3:
            return [
                    ''.join([i,j,k])
                    for i in dtol[digits[0]] 
                    for j in dtol[digits[1]]
                    for k in dtol[digits[2]]
                ]
        elif n == 4:
            return [
                    ''.join([i,j,k,l])
                    for i in dtol[digits[0]] 
                    for j in dtol[digits[1]]
                    for k in dtol[digits[2]]
                    for l in dtol[digits[3]]
                ]

RT -> O(4^n*n)
