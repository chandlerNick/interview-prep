# 216. Combination Sum III

class Solution:
    def backtrack(self, remain, k, start, path, result):
        if len(path) == k:
            if remain == 0:
                # found a valid combo
                result.append(list(path))
            return
        
        for i in range(start, 10):
            if i > remain:  # Optimization: no point in continuing if i is too large
                break
            path.append(i)
            self.backtrack(remain - i, k, i + 1, path, result)
            path.pop()
    
    def combinationSum3(self, k: int, n: int) -> List[List[int]]:
        ints = [1,2,3,4,5,6,7,8,9]
        if sum(ints[:k]) > n:
            return []
        if sum(ints[9-k:9]) < n:
            return []
        
        result = []
        self.backtrack(n, k, 1, [], result)
        return result

RT -> O(nCr(9,k)k)
Space -> O(k)
