# 215. Kth Largest Element in an Array

import heapq

class Solution:
    def findKthLargest(self, nums: List[int], k: int) -> int:
        nums = [-x for x in nums]
        heapq.heapify(nums)

        for i in range(k-1):
            heapq.heappop(nums)
        return -heapq.heappop(nums)

# RT -> O(n + klog(n))
# Space -> O(1)
