# 2336. Smallest Number in Infinite Set

import heapq

class SmallestInfiniteSet:
    def __init__(self):
        # The smallest number that has NEVER been touched
        self.curr = 1
        # Numbers that were popped but then added back
        self.added_back_heap = []
        # A set to keep track of what's in the heap (for O(1) lookup)
        self.added_back_set = set()

    def popSmallest(self) -> int:
        # Priority 1: Check the bucket of "added back" numbers
        if self.added_back_heap:
            smallest = heapq.heappop(self.added_back_heap)
            self.added_back_set.remove(smallest)
            return smallest
        
        # Priority 2: Take the next number from the infinite sequence
        smallest = self.curr
        self.curr += 1
        return smallest

    def addBack(self, num: int) -> None:
        # Only add back if it's smaller than curr (meaning it's currently missing)
        # and ensure it's not already in our heap/set (to avoid duplicates)
        if num < self.curr and num not in self.added_back_set:
            heapq.heappush(self.added_back_heap, num)
            self.added_back_set.add(num)


# RT -> O(logn)
# space -> O(n)
