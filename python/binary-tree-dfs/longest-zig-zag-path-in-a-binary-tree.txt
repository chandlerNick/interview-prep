# 1372. Longest ZigZag Path in a Binary Tree

# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def longestZigZag(self, root: Optional[TreeNode]) -> int:
        self.max_len = 0
        
        def dfs(node, go_left, curr_len):
            if not node:
                return
            
            self.max_len = max(self.max_len, curr_len)
            
            if go_left:
                # To continue the zigzag, move left and flip direction
                dfs(node.left, False, curr_len + 1)
                # Or start a new zigzag by moving right
                dfs(node.right, True, 1)
            else:
                # To continue the zigzag, move right and flip direction
                dfs(node.right, True, curr_len + 1)
                # Or start a new zigzag by moving left
                dfs(node.left, False, 1)
        
        # Start DFS from both directions at the root
        dfs(root.left, False, 1)
        dfs(root.right, True, 1)
        
        return self.max_len

# RT -> O(n)
# Space -> O(h)
