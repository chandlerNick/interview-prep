# 1137. Nth Tribonacci Number

class Solution:
    def tribonacci(self, n: int) -> int:
        tri_arr = [0, 1, 1]
        if n == 0:
            return tri_arr[0]
        if n == 1 or n == 2:
            return tri_arr[1]
        
        for i in range(2, n):
            new_num = sum(tri_arr)
            tri_arr = tri_arr[1:] + [new_num]
        
        return tri_arr[2]

Runtime -> O(n)
space -> O(1)
