# 198. House Robber

class Solution:
    def rob(self, nums: List[int]) -> int:
        # constraint: i, i+1 (equiv i-1, i) cannot be robbed
        # recurrence f(i) = max(f(i - 1), nums[i] + f(i - 2))
        # f(0) = nums[0], f(1) = max(nums[0], nums[1])
        n = len(nums)
        if n < 2:
            return max(nums)
        
        f_vals = [nums[0], max(nums[0], nums[1])]
        
        for i in range(2, n):
            f_vals.append(max(f_vals[-1], nums[i] + f_vals[-2]))
            f_vals.pop(0)
        return f_vals[-1]


Time: O(n)
Space: O(1)
