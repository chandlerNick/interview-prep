# 790. Domino and Tromino Tiling

class Solution:
    def numTilings(self, n: int) -> int:
        if n == 1:
            return 1
        elif n == 2:
            return 2
        elif n == 3:
            return 5

        # f(n) = 2f(n-1) + f(n-3)
        f_n_1 = 5
        f_n_2 = 2
        f_n_3 = 1
        f_n = 2 * f_n_1 + f_n_3
            
        for i in range(4, n):
            f_n_3 = f_n_2
            f_n_2 = f_n_1
            f_n_1 = f_n
            f_n = 2 * f_n_1 + f_n_3

        return f_n % (10**9 + 7)

RT -> O(n)
Space -> O(1)
