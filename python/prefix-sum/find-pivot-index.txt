# 724. Find Pivot Index

# RT - O(n), Space - O(1)

class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        left_sum = 0
        right_sum = sum(nums[1:])
        n = len(nums)

        if left_sum == right_sum:
            return 0

        for i in range(1,n):
            # Compute left and right sums
            left_sum += nums[i-1]
            right_sum -= nums[i]
    
            # Check that we have a pivot index
            if left_sum == right_sum:
                return i

        return -1
