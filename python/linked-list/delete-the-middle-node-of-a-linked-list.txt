# 2095. Delete the Middle Node of a Linked List

# Definition for singly-linked list.
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

class Solution:
    def deleteMiddle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if head.next is None:
            return None
        
        # Get the length -- could be another method...
        node = head
        counter = 1
        while node.next != None:
            node = node.next
            counter += 1
        
        # determine number of middle node
        middle_idx = counter // 2

        # Walk to middle node
        counter = 1
        node = head
        while counter < middle_idx and node.next:
            node = node.next
            counter += 1
        
        if node is None or node.next is None:
            return head

        # Skip the middle node to "delete it"
        # set the next to node.next.next, set node.next to null (or ignore it! we dont have to clear the memory!)
        node.next = node.next.next
        return head

# RT -> O(n)
# Space -> O(1)

# Fast soln (constant smaller):

class Solution:
    def deleteMiddle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        if not head or not head.next:
            return None
        
        slow = head
        fast = head.next.next

        # Move fast 2 at a time.
        while fast and fast.next:
            slow = slow.next
            fast = fast.next.next
            
        # slow is now at node (n/2 - 1), delete slow.next
        slow.next = slow.next.next
        return head
