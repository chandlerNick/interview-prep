# 2352. Equal Row and Column Pairs

import numpy as np

class Solution:
    def equalPairs(self, grid: List[List[int]]) -> int:
        n = len(grid)
        num_equal = 0
        grid = np.array(grid)

        for i in range(n):
            for j in range(n):
                if np.array_equal(grid[i, :], grid[:, j]):
                    num_equal += 1

        return num_equal


# RT -> O(n^3)
# Space -> O(n^2)
