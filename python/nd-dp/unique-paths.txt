# 62. Unique Paths

Combinatorics:

import math

class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        return int(math.factorial(m + n - 2) / (math.factorial(m-1) * math.factorial(n-1)))


Dynamic programming:

class Solution:
    def uniquePaths(self, m: int, n: int) -> int:
        # Initialize a 1D array representing the columns of a single row.
        # Base case: There is exactly 1 way to reach any cell in the first row.
        row = [1] * n
        
        # Iterate through the remaining rows (from row 1 to m-1)
        for _ in range(1, m):
            # The first element row[0] remains 1 because there's only 1 way to go straight down the first column.
            for c in range(1, n):
                # The new value at row[c] is the sum of:
                #   row[c]     -> the value from the row directly above (before update)
                #   row[c-1]   -> the value from the cell to the left (just updated)
                row[c] += row[c-1]
                
        return row[-1]
