# 714. Best Time to Buy and Sell Stock with Transaction Fee

class Solution:
    def maxProfit(self, prices: List[int], fee: int) -> int:
        hold = -prices[0]
        free = 0

        for i in range(len(prices)):
            temp_hold = hold
            hold = max(hold, free - prices[i])
            free = max(free, temp_hold + prices[i] - fee)

        return free

RT -> O(n)
Space -> O(1)
