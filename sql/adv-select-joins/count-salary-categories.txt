# 1907. Count Salary Categories

SELECT 'Low Salary' AS category, COUNT(income) AS accounts_count
FROM Accounts
WHERE income < 20000

UNION ALL

SELECT 'Average Salary', COUNT(income)
FROM Accounts
WHERE income >= 20000 AND income <= 50000

UNION ALL

SELECT 'High Salary', COUNT(income)
FROM Accounts
WHERE income > 50000;
