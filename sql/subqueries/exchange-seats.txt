# 626. Exchange Seats

SELECT 
    CASE 
        -- If id is odd and it's the last row, keep the id the same
        WHEN MOD(id, 2) != 0 AND id = (SELECT COUNT(*) FROM Seat) THEN id
        -- If id is odd and not the last row, increment it
        WHEN MOD(id, 2) != 0 THEN id + 1
        -- If id is even, decrement it
        ELSE id - 1
    END AS id,
    student
FROM Seat
ORDER BY id ASC;
