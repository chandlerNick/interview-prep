# 185. Department Top Three Salaries

SELECT 
    d.Name AS Department, 
    e.Name AS Employee, 
    e.Salary AS Salary
FROM Employee e
JOIN Department d ON e.DepartmentId = d.Id
WHERE 3 > (
    SELECT COUNT(DISTINCT e2.Salary)
    FROM Employee e2
    WHERE e2.DepartmentId = e.DepartmentId 
      AND e2.Salary > e.Salary
);
