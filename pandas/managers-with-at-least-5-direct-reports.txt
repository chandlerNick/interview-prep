# 570. Managers with at Least 5 Direct Reports

import pandas as pd

def find_managers(employee: pd.DataFrame) -> pd.DataFrame:
    manager_ids = employee.groupby(by="managerId", as_index=False).count()[employee.groupby(by="managerId", as_index=False).count()["department"] >= 5]["managerId"]
    return employee[employee.id.isin(manager_ids)][["name"]]
    
# Faster

import pandas as pd
def find_managers(employee: pd.DataFrame) -> pd.DataFrame:
    managers = employee.groupby('managerId').size()
    managers = managers[managers >= 5].index
    
    return employee[employee['id'].isin(managers)][['name']]
