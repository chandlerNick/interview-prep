# 1484. Group Sold Products By The Date

import pandas as pd

def categorize_products(activities: pd.DataFrame) -> pd.DataFrame:
    groups = pd.DataFrame(activities.groupby('sell_date', as_index=False)['product'].unique())
    groups["num_sold"] = groups["product"].apply(lambda x: len(x))
    groups["products"] = groups["product"].apply(lambda x: ",".join(sorted(x)))
    groups.drop(columns = ["product"], inplace=True)
    return groups
