# 1693. Daily Leads and Partners

import pandas as pd

def daily_leads_and_partners(daily_sales: pd.DataFrame) -> pd.DataFrame:
    uni = daily_sales.groupby(by=["date_id", "make_name"], as_index=False).nunique()
    uni.columns = ["date_id", "make_name", "unique_leads", "unique_partners"]
    return uni
