# 1907. Count Salary Categories

import pandas as pd

def count_salary_categories(accounts: pd.DataFrame) -> pd.DataFrame:
    low_salary = (accounts.income < 20000).sum()
    mid_salary = ((accounts.income <= 50000) * (accounts.income >= 20000)).sum()
    high_salary = (accounts.income > 50000).sum()
    return pd.DataFrame({"category": ["Low Salary", "Average Salary", "High Salary"], "accounts_count": [low_salary, mid_salary, high_salary]})
